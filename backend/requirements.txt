# ============================================================
# GenAI DocQA Platform — Python Dependencies
# ============================================================
# All versions are PINNED with == for deterministic builds
# This means every machine installs the exact same versions
# Never write just "fastapi" — always "fastapi==x.y.z"
#
# Update versions carefully and test before committing
# ============================================================


# ============================================================
# GROUP 1 — API Framework
# FastAPI + the server that runs it + file upload support
# ============================================================

# The web framework — handles routes, validation, WebSocket, SSE
fastapi==0.115.5

# ASGI server that actually runs FastAPI on a port
# [standard] adds: websockets, httptools, uvloop (faster event loop)
uvicorn[standard]==0.34.0

# Required for file uploads (multipart/form-data)
# Without this, POST /upload silently returns 422 — hard to debug
python-multipart==0.0.20

# Async file I/O — regular open() blocks the event loop
# Used when saving uploaded files to disk
aiofiles==24.1.0


# ============================================================
# GROUP 2 — Data Validation & Settings
# Pydantic validates all data in and out of the API
# ============================================================

# Data validation library — request bodies, response schemas
# v2 is a complete rewrite — much faster than v1
pydantic==2.10.4

# Same as pydantic but for environment variables
# Reads .env file, validates types, fails fast at startup
pydantic-settings==2.7.0


# ============================================================
# GROUP 3 — Database
# SQLAlchemy (ORM) → asyncpg (driver) → PostgreSQL
# ============================================================

# The ORM — write Python classes instead of raw SQL
# Version 2.0 added true async support (1.x async was a hack)
sqlalchemy==2.0.36

# Database migration tool — git for your database schema
# Tracks every schema change, makes them reversible
alembic==1.14.0

# Async PostgreSQL driver — SQLAlchemy uses this under the hood
# The chain: SQLAlchemy → asyncpg → PostgreSQL
asyncpg==0.30.0

# Adds Vector(1536) column type for pgvector extension
# Required for semantic search / embedding storage
pgvector==0.3.6


# ============================================================
# GROUP 4 — Auth & Security
# JWT tokens + password hashing + API key encryption
# ============================================================

# JWT creation and verification
# [cryptography] enables RS256/ES256 algorithms (stronger than HS256)
python-jose[cryptography]==3.3.0

# Password hashing — NEVER store plain text passwords
# [bcrypt] is the hashing algorithm — industry standard, intentionally slow
passlib[bcrypt]==1.7.4

# AES-256-GCM encryption for LLM API keys stored in DB
# Used in BYOK (Bring Your Own Key) feature
cryptography==44.0.0


# ============================================================
# GROUP 5 — HTTP Client & Caching
# Making external API calls + Redis + retry logic
# ============================================================

# Async HTTP client — for calling LLM APIs, Tavily search, etc.
# requests library is synchronous (blocks) — httpx is async
httpx==0.28.1

# Async Redis client — rate limiting, query cache, embedding cache
redis==5.2.1

# Retry logic with exponential backoff
# LLM API fails → wait 2s → retry → wait 4s → retry → give up
tenacity==9.0.0
